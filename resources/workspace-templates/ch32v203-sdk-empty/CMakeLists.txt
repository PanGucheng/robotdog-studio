cmake_minimum_required(VERSION 3.24)
set(CMAKE_TRY_COMPILE_PLATFORM_VARIABLES ROBOTDOG_TOOLCHAIN_ROOT)
project(RobotDogFirmware C ASM)
set(ROBOTDOG_OUTPUT_DIR "${CMAKE_BINARY_DIR}/out" CACHE PATH "Firmware output")
file(MAKE_DIRECTORY "${ROBOTDOG_OUTPUT_DIR}")
file(GLOB_RECURSE SDK_SOURCES CONFIGURE_DEPENDS Peripheral/src/*.c)
file(GLOB_RECURSE APP_SOURCES CONFIGURE_DEPENDS App/Src/*.c)
add_executable(RobotDogFirmware Startup/startup_ch32v20x_D6.S Core/core_riscv.c Debug/debug.c User/system_ch32v20x.c User/ch32v20x_it.c ${SDK_SOURCES} ${APP_SOURCES})
target_include_directories(RobotDogFirmware PRIVATE Core Debug User Peripheral/inc App/Inc App/Src)
target_compile_options(RobotDogFirmware PRIVATE -march=rv32imac -mabi=ilp32 -mcmodel=medlow -Os -ffunction-sections -fdata-sections -Wall -Werror=implicit-function-declaration)
target_link_options(RobotDogFirmware PRIVATE -march=rv32imac -mabi=ilp32 -mcmodel=medlow -nostartfiles --specs=nano.specs --specs=nosys.specs -T "${CMAKE_SOURCE_DIR}/Ld/Link.ld" -Wl,--gc-sections "-Wl,-Map=${ROBOTDOG_OUTPUT_DIR}/RobotDog.map")
set_target_properties(RobotDogFirmware PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${ROBOTDOG_OUTPUT_DIR}" OUTPUT_NAME "RobotDog" SUFFIX ".elf")
add_custom_command(TARGET RobotDogFirmware POST_BUILD COMMAND "${CMAKE_OBJCOPY}" -O ihex "$<TARGET_FILE:RobotDogFirmware>" "${ROBOTDOG_OUTPUT_DIR}/RobotDog.hex" COMMAND "${CMAKE_OBJCOPY}" -O binary "$<TARGET_FILE:RobotDogFirmware>" "${ROBOTDOG_OUTPUT_DIR}/RobotDog.bin" VERBATIM)
